# AI（Illustrator）轉 SVG 工具 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 hd-toolkit 新增一個工具，把內含多個 artboard 的 `.ai` / `.pdf` 逐頁拆成獨立向量 SVG，於瀏覽器預覽、勾選、逐頁命名後單檔或打包下載。

**Architecture:** 沿用 pdf2jpg 的後端骨架，但回傳 JSON（每頁一段 SVG）而非 ZIP，讓 SVG「預覽即成品」一物二用；後端用 `subprocess` 呼叫 poppler 的 `pdfinfo` / `pdftocairo -svg`（無狀態）；前端負責預覽、勾選、命名、以 JSZip 打包（後端保持無狀態）。

**Tech Stack:** FastAPI、poppler（`pdfinfo` / `pdftocairo`，已因 pdf2jpg 存在）、pytest + Starlette TestClient（新引入）、純 HTML/CSS/JS 前端、JSZip（cdnjs）。

## Global Constraints

- 前端無框架，純 HTML/CSS/JS；套用 `static/shared/app.css`、`theme.js`、`settings.js`。
- 視覺一律遵守 `DESIGN.md`（Quiet Utility、暖琥珀 accent、亮暗雙模式、`var(--...)` tokens、圖示 inline SVG 線條圖、**無 Emoji**）。
- 第三方前端庫從 cdnjs 引入，帶 `integrity="sha512-…"`（SRI，防 CDN 被竄改）+ `crossorigin="anonymous" referrerpolicy="no-referrer"`，pin 明確版本（比照 invoice-stamp 的 pdf.js 3.11.174，但額外加 SRI 把 baseline 拉高）。
- 後端 `subprocess` 一律用**參數陣列**，**絕不** `shell=True`、絕不字串拼接檔名；每次呼叫設 `timeout=30`。
- 檔案界限：大小上限 20MB、頁數上限 200；副檔名（`.ai`/`.pdf`）+ magic bytes（`%PDF`）雙重驗證。
- 前端預覽 SVG 一律用 `<img src="blob:…">`（image context 不執行 SVG 內 `<script>`），不用 `innerHTML` 內嵌。
- 錯誤訊息繁中、對一般使用者友善，不外洩 traceback / 內部路徑。
- Commit 訊息用專案風格：`ai2svg: <英文簡述>`，**不加** Claude 尾註。

```python
"""pytest 共用設定：確保測試能 import 專案根模組（main / app），並提供
測試用的多頁向量 PDF 產生器（無外部依賴、確定性、非客戶資產）。"""

import sys
from pathlib import Path

ROOT = Path(__file__).parent
if str(ROOT) not in sys.path:
    sys.path.insert(0, str(ROOT))


def make_pdf(n_pages: int) -> bytes:
    """手寫最小多頁 PDF，每頁一個填色矩形（向量繪圖），顏色隨頁不同。"""
    objs = []
    kids = [3 + i * 2 for i in range(n_pages)]
    objs.append(b"<< /Type /Catalog /Pages 2 0 R >>")
    kids_ref = " ".join(f"{k} 0 R" for k in kids).encode()
    objs.append(
        b"<< /Type /Pages /Kids [" + kids_ref + b"] /Count "
        + str(n_pages).encode() + b" >>"
    )
    for i in range(n_pages):
        content_obj = 3 + i * 2 + 1
        r = (i * 0.3) % 1.0
        stream = f"{r:.2f} 0.20 0.60 rg 100 100 300 300 re f".encode()
        objs.append(
            b"<< /Type /Page /Parent 2 0 R /MediaBox [0 0 500 500] "
            b"/Contents " + str(content_obj).encode() + b" 0 R /Resources << >> >>"
        )
        objs.append(
            b"<< /Length " + str(len(stream)).encode() + b" >>\nstream\n"
            + stream + b"\nendstream"
        )
    out = bytearray(b"%PDF-1.4\n")
    offsets = []
    for idx, body in enumerate(objs, start=1):
        offsets.append(len(out))
        out += f"{idx} 0 obj\n".encode() + body + b"\nendobj\n"
    xref_pos = len(out)
    out += f"xref\n0 {len(objs) + 1}\n".encode()
    out += b"0000000000 65535 f \n"
    for off in offsets:
        out += f"{off:010d} 00000 n \n".encode()
    out += b"trailer\n<< /Size " + str(len(objs) + 1).encode() + b" /Root 1 0 R >>\n"
    out += b"startxref\n" + str(xref_pos).encode() + b"\n%%EOF"
    return bytes(out)
```

```python
"""AI（Adobe Illustrator）/ PDF 轉 SVG 工具 — 後端 router

把上傳的 .ai / .pdf 逐頁（artboard）用 poppler 的 pdftocairo 轉成獨立向量
SVG，回傳 JSON（每頁一段 SVG）。需要系統套件 poppler（pdfinfo / pdftocairo，
已因 pdf2jpg 而存在）。
"""

import subprocess
import tempfile
from pathlib import Path

from fastapi import APIRouter, File, HTTPException, UploadFile

router = APIRouter(tags=["ai2svg"])

MAX_BYTES = 20 * 1024 * 1024  # 20MB
MAX_PAGES = 200
TIMEOUT = 30  # 秒


def _page_count(pdf_path: str) -> int:
    """用 pdfinfo 取頁數。"""
    try:
        proc = subprocess.run(
            ["pdfinfo", pdf_path],
            capture_output=True, text=True, timeout=TIMEOUT, check=True,
        )
    except FileNotFoundError:
        raise HTTPException(status_code=500, detail="伺服器缺少 poppler（pdfinfo），無法處理")
    except subprocess.TimeoutExpired:
        raise HTTPException(status_code=500, detail="讀取頁數逾時")
    except subprocess.CalledProcessError:
        raise HTTPException(status_code=400, detail="無法讀取頁數，檔案可能已損壞或非 PDF 相容格式")
    for line in proc.stdout.splitlines():
        if line.startswith("Pages:"):
            return int(line.split(":", 1)[1].strip())
    raise HTTPException(status_code=400, detail="無法判讀頁數")


def _page_to_svg(pdf_path: str, page: int, out_path: str) -> str:
    """用 pdftocairo -svg 把單頁轉成 SVG，回傳 SVG 文字。"""
    try:
        subprocess.run(
            ["pdftocairo", "-svg", "-f", str(page), "-l", str(page), pdf_path, out_path],
            capture_output=True, timeout=TIMEOUT, check=True,
        )
    except FileNotFoundError:
        raise HTTPException(status_code=500, detail="伺服器缺少 poppler（pdftocairo），無法處理")
    except subprocess.TimeoutExpired:
        raise HTTPException(status_code=500, detail=f"第 {page} 頁轉換逾時")
    except subprocess.CalledProcessError:
        raise HTTPException(status_code=500, detail=f"第 {page} 頁轉換失敗")
    return Path(out_path).read_text(encoding="utf-8")


@router.post("/api/ai2svg/convert")
async def convert(file: UploadFile = File(..., description=".ai 或 .pdf 檔案")):
    """把 .ai / .pdf 逐頁轉成 SVG，回傳 JSON。"""
    if not file.filename:
        raise HTTPException(status_code=400, detail="未提供檔案名稱")
    ext = Path(file.filename).suffix.lower()
    if ext not in (".ai", ".pdf"):
        raise HTTPException(status_code=400, detail="只接受 .ai 或 .pdf 檔案")

    data = await file.read()
    if len(data) == 0:
        raise HTTPException(status_code=400, detail="檔案為空")
    if len(data) > MAX_BYTES:
        raise HTTPException(status_code=400, detail="檔案超過 20MB 上限")
    if not data.startswith(b"%PDF"):
        raise HTTPException(
            status_code=400,
            detail="此檔非 PDF 相容格式（舊版 .ai 需先在 Illustrator 另存為 PDF 相容）",
        )

    with tempfile.TemporaryDirectory() as tmp:
        src = Path(tmp) / "src.pdf"
        src.write_bytes(data)
        count = _page_count(str(src))
        if count > MAX_PAGES:
            raise HTTPException(status_code=400, detail=f"頁數超過上限（{MAX_PAGES} 頁）")
        pages = []
        for p in range(1, count + 1):
            out = Path(tmp) / f"p{p}.svg"
            pages.append({"index": p, "svg": _page_to_svg(str(src), p, str(out))})

    return {"filename": Path(file.filename).stem, "page_count": count, "pages": pages}
```

```html
      function showError(msg) { errorText.textContent = msg; errorMsg.classList.add('is-shown'); }
      function hideError() { errorMsg.classList.remove('is-shown'); }

      function sanitizeName(s) {
        return (s || '').replace(/[\/\\:*?"<>|]/g, '').trim();
      }
      function pad2(n) { return (n < 10 ? '0' : '') + n; }

      function handleFile(file) {
        var name = (file && file.name || '').toLowerCase();
        if (!file || !(name.endsWith('.ai') || name.endsWith('.pdf'))) {
          showError('請選擇 .ai 或 .pdf 檔案。');
          return;
        }
        selectedFile = file;
        fileName.textContent = file.name;
        fileName.classList.add('is-shown');
        convertBtn.disabled = false;
        convertBtn.textContent = '開始轉換';
        hideError();
      }

      dropzone.addEventListener('click', function () { fileInput.click(); });
      dropzone.addEventListener('dragover', function (e) { e.preventDefault(); dropzone.classList.add('is-dragover'); });
      dropzone.addEventListener('dragleave', function (e) { e.preventDefault(); dropzone.classList.remove('is-dragover'); });
      dropzone.addEventListener('drop', function (e) {
        e.preventDefault();
        dropzone.classList.remove('is-dragover');
        if (e.dataTransfer.files.length) handleFile(e.dataTransfer.files[0]);
      });
      fileInput.addEventListener('change', function (e) {
        if (e.target.files.length) handleFile(e.target.files[0]);
      });

      function updateSelCount() {
        var checks = svgGrid.querySelectorAll('input[type=checkbox]');
        var n = 0;
        checks.forEach(function (c) { if (c.checked) n++; });
        selCount.textContent = n + ' / ' + checks.length + ' 已選';
        downloadZipBtn.disabled = n === 0;
      }

      function renderPages() {
        svgGrid.innerHTML = '';
        pages.forEach(function (pg, i) {
          var item = document.createElement('div');
          item.className = 'svg-item is-selected';

          var preview = document.createElement('div');
          preview.className = 'svg-item__preview';
          var img = document.createElement('img');
          img.src = pg.blobUrl;   // SVG 以圖片載入，不執行內部 script
          img.alt = pg.name;
          preview.appendChild(img);

          var body = document.createElement('div');
          body.className = 'svg-item__body';

          var row1 = document.createElement('div');
          row1.className = 'svg-item__row';
          var label = document.createElement('label');
          var cb = document.createElement('input');
          cb.type = 'checkbox';
          cb.checked = true;
          cb.addEventListener('change', function () {
            item.classList.toggle('is-selected', cb.checked);
            updateSelCount();
          });
          var labelText = document.createElement('span');
          labelText.textContent = '第 ' + pg.index + ' 頁';
          label.appendChild(cb);
          label.appendChild(labelText);
          var dlOne = document.createElement('button');
          dlOne.type = 'button';
          dlOne.className = 'btn btn--ghost';
          dlOne.textContent = '下載此頁';
          dlOne.addEventListener('click', function () { downloadOne(pg); });
          row1.appendChild(label);
          var spacer = document.createElement('span');
          spacer.style.flex = '1';
          row1.appendChild(spacer);
          row1.appendChild(dlOne);

          var row2 = document.createElement('div');
          row2.className = 'svg-item__row';
          var nameInput = document.createElement('input');
          nameInput.type = 'text';
          nameInput.className = 'text-field svg-item__name';
          nameInput.value = pg.name;
          nameInput.addEventListener('input', function () { pg.name = nameInput.value; });
          var ext = document.createElement('span');
          ext.className = 'svg-item__ext';
          ext.textContent = '.svg';
          row2.appendChild(nameInput);
          row2.appendChild(ext);

          body.appendChild(row1);
          body.appendChild(row2);
          item.appendChild(preview);
          item.appendChild(body);
          svgGrid.appendChild(item);

          pg._checkbox = cb;
        });
        updateSelCount();
      }

      function fileNameFor(pg) {
        var n = sanitizeName(pg.name) || (baseName + '-' + pad2(pg.index));
        return n + '.svg';
      }

      function downloadOne(pg) {
        var blob = new Blob([pg.svg], { type: 'image/svg+xml' });
        var url = URL.createObjectURL(blob);
        var a = document.createElement('a');
        a.href = url;
        a.download = fileNameFor(pg);
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);
        URL.revokeObjectURL(url);
      }

      selectAllBtn.addEventListener('click', function () {
        svgGrid.querySelectorAll('input[type=checkbox]').forEach(function (c) {
          c.checked = true; c.dispatchEvent(new Event('change'));
        });
      });
      selectNoneBtn.addEventListener('click', function () {
        svgGrid.querySelectorAll('input[type=checkbox]').forEach(function (c) {
          c.checked = false; c.dispatchEvent(new Event('change'));
        });
      });

      downloadZipBtn.addEventListener('click', async function () {
        var chosen = pages.filter(function (pg) { return pg._checkbox.checked; });
        if (!chosen.length) return;
        var zip = new JSZip();
        var used = {};
        chosen.forEach(function (pg) {
          var fn = fileNameFor(pg);
          if (used[fn]) { fn = fn.replace(/\.svg$/, '') + '-' + pg.index + '.svg'; }
          used[fn] = true;
          zip.file(fn, pg.svg);
        });
        var blob = await zip.generateAsync({ type: 'blob' });
        var url = URL.createObjectURL(blob);
        var a = document.createElement('a');
        a.href = url;
        a.download = baseName + '-svg.zip';
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);
        URL.revokeObjectURL(url);
      });

      convertBtn.addEventListener('click', async function () {
        if (!selectedFile) return;
        hideError();
        resultCard.hidden = true;
        progress.classList.add('is-shown');
        progressBar.style.width = '20%';
        progressStatus.textContent = '上傳中…';
        convertBtn.disabled = true;

        var form = new FormData();
        form.append('file', selectedFile);

        try {
          var res = await fetch('/api/ai2svg/convert', { method: 'POST', body: form });
          progressBar.style.width = '70%';
          progressStatus.textContent = '處理中…';
          if (!res.ok) {
            var err; try { err = await res.json(); } catch (e) { err = {}; }
            throw new Error(err.detail || ('轉換失敗（HTTP ' + res.status + '）'));
          }
          var data = await res.json();
          baseName = sanitizeName(data.filename) || 'export';
          pages = data.pages.map(function (p) {
            var blob = new Blob([p.svg], { type: 'image/svg+xml' });
            return {
              index: p.index,
              svg: p.svg,
              blobUrl: URL.createObjectURL(blob),
              name: baseName + '-' + pad2(p.index)
            };
          });
          renderPages();
          progressBar.style.width = '100%';
          progressStatus.textContent = '完成';
          setTimeout(function () {
            progress.classList.remove('is-shown');
            resultCard.hidden = false;
          }, 350);
        } catch (e) {
          showError(e.message || '轉換失敗。');
          progress.classList.remove('is-shown');
        } finally {
          convertBtn.disabled = false;
        }
      });
    })();
  </script>
</body>
</html>
```
